### Specification
`src/index.ts`:
```ts
import type { ExtensionAPI } from "@earendil-works/pi-coding-agent";
import { fauxAssistantMessage, fauxProvider, fauxText, fauxToolCall } from "@earendil-works/pi-ai";
import { readFileSync } from "node:fs";

type Step = { text?: string; tool_calls?: { name: string; arguments: Record<string, unknown> }[]; error?: string };

export default function (pi: ExtensionAPI) {
  const path = process.env.SI_FAUX_SCRIPT;
  if (!path) return;                                   // inert unless a script is given
  const script = JSON.parse(readFileSync(path, "utf8")) as { model?: string; steps: Step[] };
  const faux = fauxProvider({ provider: "si-faux", api: "si-faux-api" /* models: see below */ });
  const responses = script.steps.map((s) =>
    s.error
      ? fauxAssistantMessage("", { stopReason: "error", errorMessage: s.error })
      : fauxAssistantMessage(
          [...(s.text ? [fauxText(s.text)] : []), ...(s.tool_calls ?? []).map((c) => fauxToolCall(c.name, c.arguments))],
          { stopReason: s.tool_calls?.length ? "toolUse" : "stop" },
        ),
  );
  responses.push(fauxAssistantMessage("", { stopReason: "error", errorMessage: "si-faux: script exhausted" }));
  faux.setResponses(responses);
  pi.registerProvider(faux.provider.id, {
    api: faux.api, baseUrl: faux.getModel().baseUrl, apiKey: "si-faux",
    models: faux.models, streamSimple: faux.provider.streamSimple,
  });
}
```
- Check `FauxModelDefinition` in `faux.ts` and pass `models: [{ id: script.model ?? "scripted", ... }]` so the model id is predictable; if the option shape differs, use the default model id and expose it by logging `si-faux model: <id>` to stderr once.
- Verify the exact `stopReason` value for tool calls in Pi's `AssistantMessage` type (`"toolUse"` in current Pi); adjust if different.
- `@earendil-works/pi-ai` must resolve inside the extension: it is installed locally by `npm ci` in `testing/pi-faux`.

### Tests
`test/si/faux_smoke_test.exs` (`@tag :pi`), using the test toolchain from `Si.Test.Toolchain.ensure!/0` and a raw Erlang Port (T03.04/T03.05 don't exist yet): start `pi --mode rpc --no-session --no-extensions -e testing/pi-faux --provider si-faux --model scripted` with the env from `session_opts/1` and a one-step script `[%{"text" => "hello from faux"}]`; write `{"id":"1","type":"prompt","message":"hi"}\n`; read stdout lines until an `agent_settled` event; assert a `message_end` (or equivalent assistant message event) contains `hello from faux`; close the port.
